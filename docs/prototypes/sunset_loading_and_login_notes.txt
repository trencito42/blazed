    <script>
        // Logica pentru Tab-uri
        function switchTab(target) {
            // Butoane
            const tabs = document.querySelectorAll('.tab-btn');
            tabs.forEach(tab => tab.classList.remove('active'));
            event.target.classList.add('active');

            // Formulare
            document.getElementById('form-login').classList.remove('active');
            document.getElementById('form-register').classList.remove('active');
            
            document.getElementById('form-' + target).classList.add('active');
        }

        // Simulare Loading Screen (Intrarea în oraș)
        function enterCity() {
            const authPanel = document.getElementById('auth-panel');
            const loadingScreen = document.getElementById('loading-screen');
            const pctText = document.getElementById('loading-pct');
            const barFill = document.getElementById('loading-bar');
            const statusText = document.getElementById('loading-status');

            // 1. Ascunde panoul de auth (glisare spre stânga)
            authPanel.classList.add('hidden');

            // 2. Afișează ecranul de loading după un scurt delay
            setTimeout(() => {
                loadingScreen.classList.add('active');
                
                // 3. Simulare progres încărcare resurse FiveM
                let progress = 0;
                const totalTime = 8000; // 8 secunde durata simulată
                const intervalTime = 50; 
                const increment = 100 / (totalTime / intervalTime);

                const statusMessages = [
                    "Connecting to server instance...",
                    "Downloading map assets...",
                    "Loading Sunset Custom Vehicles...",
                    "Initializing UI Systems...",
                    "Fetching character data...",
                    "Synchronizing weather protocols...",
                    "Awaiting game state..."
                ];

                const loadingInterval = setInterval(() => {
                    progress += increment;
                    
                    // Adaugă variație random (uneori se blochează puțin cum e la FiveM)
                    if(Math.random() > 0.8) progress -= increment * 0.8;

                    if (progress >= 100) {
                        progress = 100;
                        clearInterval(loadingInterval);
                        statusText.innerText = "Entering Los Santos...";
                        
                        // Aici s-ar face hide complet in LUA
                        setTimeout(() => {
                            document.body.style.opacity = '0'; // Fade out complet
                        }, 1000);
                    }

                    // Update UI
                    pctText.innerText = Math.floor(progress) + "%";
                    barFill.style.width = progress + "%";

                    // Schimbă textul la anumite procente
                    const msgIndex = Math.floor((progress / 100) * statusMessages.length);
                    if(statusMessages[msgIndex]) {
                        statusText.innerText = statusMessages[msgIndex];
                    }

                }, intervalTime);

            }, 500); // Așteaptă să iasă panoul

            /*
            Pentru implementare NUI în FiveM:
            Când dai click pe Enter City, trimiți un post către client:
            fetch(`https://${GetParentResourceName()}/triggerLogin`, {
                method: 'POST',
                body: JSON.stringify({
                    username: usernameInput.value,
                    password: passwordInput.value
                })
            });
            */
        }
    </script>
