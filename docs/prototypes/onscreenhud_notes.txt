        // Script simplu pentru a actualiza ceasul real pentru efectul imersiv
        function updateClock() {
            const now = new Date();
            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
            
            const clockSpan = document.querySelector('#hud-clock span');
            if(clockSpan) {
                clockSpan.textContent = `${hours}:${minutes}`;
            }
        }

        // Inițializare ceas
        updateClock();
        setInterval(updateClock, 10000); // Update la fiecare 10 secunde

        // ==========================================
        // SIMULARE VITEZOMETRU (PENTRU PREVIEW)
        // ==========================================
        const speedEl = document.getElementById('hud-speed');
        const rpmEl = document.getElementById('hud-rpm');
        const gearEl = document.getElementById('hud-gear');

        let currentSpeed = 124;
        let currentRpm = 75;
        let gear = 4;
        
        // Variabile noi pentru a detecta schimbările și a declanșa animațiile
        let lastGear = 4;
        let lastSpeed = 124;

        setInterval(() => {
            // Randomizare usoara a vitezei
            currentSpeed += Math.floor(Math.random() * 7) - 3;
            if(currentSpeed < 0) currentSpeed = 0;
            if(currentSpeed > 280) currentSpeed = 280;
            
            // Calculam RPM-ul (logica falsa pt aspect)
            currentRpm = (currentSpeed % 40) * 2.5; 
            if(currentRpm < 15 && currentSpeed > 0) currentRpm = 15;
            if(currentSpeed === 0) currentRpm = 0;
            
            // Calculam Treapta de viteza (Gear)
            gear = Math.floor(currentSpeed / 40) + 1;
            if(gear > 6) gear = 6;
            if(currentSpeed === 0) gear = 'N';

            // Update Viteza + Animație
            if(speedEl && currentSpeed !== lastSpeed) {
                speedEl.textContent = currentSpeed;
                speedEl.classList.remove('anim-speed');
                void speedEl.offsetWidth; // Truc tehnic: forțează browserul să restarteze animația CSS
                speedEl.classList.add('anim-speed');
                lastSpeed = currentSpeed;
            }
            
            // Update RPM Bar
            if(rpmEl) rpmEl.style.width = currentRpm + '%';
            
            // Update Gear + Animație agresivă
            if(gearEl && gear !== lastGear) {
                gearEl.textContent = gear === 'N' ? 'N' : 'G' + gear;
                gearEl.classList.remove('anim-gear');
                void gearEl.offsetWidth; // Truc tehnic
                gearEl.classList.add('anim-gear');
                lastGear = gear;
            }
        }, 150);
