    <script>
        // Setări Pompă
        const pricePerLiter = 2.50;
        const maxTankCapacity = 65.0; // Capacitate totală a mașinii în litri
        let currentLitersInTank = 9.75; // 15% din 65L (Cât are deja mașina)
        
        let litersDispensed = 0.00;
        let totalCost = 0.00;
        
        let isHoldingKey = false;
        let pumpInterval = null;
        let isFull = false;

        // Referințe DOM
        const uiPump = document.getElementById('pump-ui');
        const uiCost = document.getElementById('ui-cost');
        const uiLiters = document.getElementById('ui-liters');
        const uiPercent = document.getElementById('ui-percent');
        const uiTankBar = document.getElementById('ui-tank-bar');
        const promptArea = document.getElementById('ui-prompt');

        // Funcția de pompare (rulează la fiecare 30ms pentru animație fluidă de cifre)
        function processRefuel() {
            if (isFull) return;

            // Viteza de pompare per frame
            const flowRate = 0.15; 
            
            litersDispensed += flowRate;
            currentLitersInTank += flowRate;
            
            // Verificare dacă s-a umplut
            if (currentLitersInTank >= maxTankCapacity) {
                currentLitersInTank = maxTankCapacity;
                litersDispensed = maxTankCapacity - (currentLitersInTank - litersDispensed); // corectie matematica fina
                finishRefuel();
            }

            // Calcul Cost
            totalCost = litersDispensed * pricePerLiter;

            // Update UI
            uiLiters.innerText = litersDispensed.toFixed(2);
            uiCost.innerText = totalCost.toFixed(2);
            
            let percent = (currentLitersInTank / maxTankCapacity) * 100;
            uiPercent.innerText = Math.floor(percent) + '%';
            uiTankBar.style.width = percent + '%';
        }

        // Când se umple 100%
        function finishRefuel() {
            isFull = true;
            stopPumping();
            uiPump.classList.remove('is-pumping');
            uiPump.classList.add('is-full');
            
            promptArea.innerHTML = `
                <div class="prompt-text">
                    <svg viewBox="0 0 24 24" width="20" height="20" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round" stroke-linejoin="round"><polyline points="20 6 9 17 4 12"/></svg>
                    REZERVOR PLIN
                </div>
            `;
            
            // Trimite NUI Callback catre LUA ca s-a terminat
            console.log(`Pumping finished. Total Cost: $${totalCost.toFixed(2)}`);
        }

        function startPumping() {
            if(isFull || isHoldingKey) return;
            isHoldingKey = true;
            uiPump.classList.add('is-pumping');
            
            promptArea.innerHTML = `
                <div class="prompt-text">
                    POMPARE ÎN CURS...
                </div>
            `;

            // Interval rapid pentru senzația de numere rulând la pompă
            pumpInterval = setInterval(processRefuel, 30);
        }

        function stopPumping() {
            if(!isHoldingKey || isFull) return;
            isHoldingKey = false;
            uiPump.classList.remove('is-pumping');
            clearInterval(pumpInterval);
            
            promptArea.innerHTML = `
                <div class="prompt-text">
                    ȚINE APĂSAT <span class="key-btn">E</span> PENTRU A CONTINUA
                </div>
            `;
        }

        // Ascultători pentru tastatură
        window.addEventListener('keydown', (e) => {
            if (e.key.toLowerCase() === 'e') {
                startPumping();
            }
        });

        window.addEventListener('keyup', (e) => {
            if (e.key.toLowerCase() === 'e') {
                stopPumping();
            }
        });

    </script>
